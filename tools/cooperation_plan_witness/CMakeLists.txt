cmake_minimum_required(VERSION 3.16)
project(cooperation_plan_witness LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(nav2_amcl REQUIRED)
find_package(nav2_smac_planner REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
add_executable(cooperation_plan_witness main.cpp)
target_compile_features(cooperation_plan_witness PRIVATE cxx_std_17)
ament_target_dependencies(cooperation_plan_witness rclcpp nav_msgs)
add_executable(cooperation_amcl_motion_probe amcl_motion_probe.cpp)
target_compile_features(cooperation_amcl_motion_probe PRIVATE cxx_std_17)
ament_target_dependencies(cooperation_amcl_motion_probe nav2_amcl)
add_executable(cooperation_lattice_cache_probe lattice_cache_probe.cpp)
target_compile_features(cooperation_lattice_cache_probe PRIVATE cxx_std_17)
ament_target_dependencies(cooperation_lattice_cache_probe rclcpp nav2_smac_planner nav2_costmap_2d)
