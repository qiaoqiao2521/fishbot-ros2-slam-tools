cmake_minimum_required(VERSION 3.16)
project(drone_ros_ack_witness)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(traj_utils REQUIRED)
find_package(quadrotor_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(drone_ros_ack_witness witness.cpp)
target_compile_features(drone_ros_ack_witness PRIVATE cxx_std_17)
ament_target_dependencies(drone_ros_ack_witness rclcpp std_msgs traj_utils quadrotor_msgs)
target_link_libraries(drone_ros_ack_witness nlohmann_json::nlohmann_json)
