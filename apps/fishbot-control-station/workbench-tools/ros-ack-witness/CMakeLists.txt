cmake_minimum_required(VERSION 3.16)
project(drone_ros_ack_witness)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(traj_utils REQUIRED)
find_package(quadrotor_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(drone_ros_ack_witness witness.cpp)
target_compile_features(drone_ros_ack_witness PRIVATE cxx_std_17)
ament_target_dependencies(drone_ros_ack_witness rclcpp std_msgs traj_utils quadrotor_msgs)
target_link_libraries(drone_ros_ack_witness nlohmann_json::nlohmann_json)

option(DRONE_BUILD_TRAJ_LIFETIME_PROBE "Build a copied upstream trajectory server for lifetime diagnosis" OFF)
set(DRONE_TRAJ_SERVER_SOURCE "" CACHE FILEPATH "Patched temporary traj_server.cpp, never the upstream source tree")
if(DRONE_BUILD_TRAJ_LIFETIME_PROBE AND EXISTS "${DRONE_TRAJ_SERVER_SOURCE}")
  find_package(bspline_opt REQUIRED)
  find_package(Eigen3 REQUIRED)
  find_package(OpenMP REQUIRED)
  find_package(nav_msgs REQUIRED)
  find_package(visualization_msgs REQUIRED)
  add_executable(drone_traj_lifetime_probe "${DRONE_TRAJ_SERVER_SOURCE}")
  target_compile_features(drone_traj_lifetime_probe PRIVATE cxx_std_17)
  ament_target_dependencies(drone_traj_lifetime_probe rclcpp std_msgs nav_msgs
    visualization_msgs traj_utils quadrotor_msgs bspline_opt)
  target_link_libraries(drone_traj_lifetime_probe Eigen3::Eigen OpenMP::OpenMP_CXX)
elseif(DRONE_BUILD_TRAJ_LIFETIME_PROBE)
  message(WARNING "No copied trajectory source supplied; only the receipt witness will be built")
endif()
